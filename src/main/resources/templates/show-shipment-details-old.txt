  <script>
    // ── helpers ──
    function openModal(id)  { document.getElementById(id).classList.add('open');    document.body.style.overflow = 'hidden'; }
    function closeModal(id) { document.getElementById(id).classList.remove('open'); document.body.style.overflow = '';       }
    // close on overlay click
    document.querySelectorAll('.modal-overlay').forEach(overlay => {
      overlay.addEventListener('click', e => { if (e.target === overlay) closeModal(overlay.id); });
    });

    // ── msg box ──
   window.addEventListener("load", function () {
	    const msg = document.getElementById("msgBox");
	
	    if (!msg) return;
	
	 	// this will tell whether to hide the msg after 10secs or to always show it
	 	// if autoHide is set to true that means, shipment status is updated successfully so we can close the message orprevent the message in red ribbon to always show & not disapper after 10secs
	 	// if autoHide is false, that means status update has failed due to some errors, so this msg will appear untill the page is refreshed or shipment details is reloaded
	    const autoHide = msg.dataset.autohide === "true";
	
	    if (autoHide) {
	        setTimeout(() => msg.style.display = "none", 10000);
	    }
    
	});

    // ── disable buttons if needed ──
    document.addEventListener("DOMContentLoaded", function () {
      var disable = document.getElementById("disableButtonActions");
      if (disable && disable.value === "true") {
        document.querySelectorAll("button:not(.modal-close-x):not(.btn-modal-cancel):not(.btn-modal-close)").forEach(btn => {
          btn.disabled = true;
          btn.title = "Action disabled — customer inactive or contract expired";
        });
      }
    });

    // ══ CANCEL MODAL ══
    const openCancelBtn = document.getElementById('openCancelModal');
    if (openCancelBtn) openCancelBtn.addEventListener('click', () => openModal('cancelModal'));

    const cancelInput   = document.getElementById('cancelReasonInput');
    const cancelConfirm = document.getElementById('cancelConfirmBtn');
    const cancelError   = document.getElementById('cancelReasonError');
    const cancelCount   = document.getElementById('cancelCharCount');

    if (cancelInput) {
      cancelInput.addEventListener('input', function () {
        const len = this.value.trim().length;
        cancelCount.textContent = `${this.value.length} / 60`;
        if (len < 6) {
          cancelError.textContent = len === 0 ? '' : `At least 6 characters required (${len} entered).`;
          cancelConfirm.disabled = true;
          cancelInput.classList.add('error');
        } else {
          cancelError.textContent = '';
          cancelConfirm.disabled = false;
          cancelInput.classList.remove('error');
        }
      });
    }

    if (cancelConfirm) {
      cancelConfirm.addEventListener('click', function () {
        document.getElementById('cancelReasonHidden').value = cancelInput.value.trim();
        document.getElementById('cancelForm').submit();
      });
    }

    // ══ AUDIT MODAL ══
    const shipmentId = /*[[${shipment.shipmentId}]]*/ '';

    function openAuditModal() {

        openModal('auditModal');

        const loading = document.getElementById('auditLoading');
        const table   = document.getElementById('auditTable');
        const tbody   = document.getElementById('auditTableBody');

        loading.style.display = 'block';
        loading.textContent   = 'Loading audit records...';

        table.style.display = 'none';
        tbody.innerHTML = '';

        const shipmentId = document.getElementById('shipmentId__1').value;

        fetch('/shipments/getShipmentAudit/' + shipmentId, {
            method: 'GET',
            headers: {
                'Accept': 'application/json'
            }
        })
        .then(response => {
            if (!response.ok) {
                throw new Error('Failed to fetch audit details');
            }
            return response.json();
        })
        .then(data => {

            loading.style.display = 'none';

            if (!data || data.length === 0) {
                loading.style.display = 'block';
                loading.textContent = 'No audit records found.';
                return;
            }

            data.forEach(row => {

                const status  = row.shipmentStatus || '—';
                const dateTime = row.eventDate || '—';
                const comment  = row.comment || '—';

                tbody.innerHTML += `
                    <tr>
                        <td>${status}</td>
                        <td>${dateTime}</td>
                        <td>${comment}</td>
                    </tr>
                `;
            });

            table.style.display = 'table';
        })
        .catch(error => {

            console.error(error);

            loading.style.display = 'block';
            loading.textContent = 'Error loading audit records.';
        });
    }

    const auditBtn         = document.getElementById('openAuditModal');
    const auditBtnTerminal = document.getElementById('openAuditModalTerminal');
    if (auditBtn)         auditBtn.addEventListener('click',         openAuditModal);
    if (auditBtnTerminal) auditBtnTerminal.addEventListener('click', openAuditModal);
  </script>
